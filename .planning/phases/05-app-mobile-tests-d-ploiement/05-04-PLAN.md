```yaml
# .github/workflows/deploy-web.yml
# Déploiement automatique de l'app web Next.js sur Vercel à chaque push sur main
# Secrets requis dans GitHub → Settings → Secrets and variables → Actions:
#   VERCEL_TOKEN: token API Vercel (vercel.com/account/tokens)
#   VERCEL_ORG_ID: trouvable dans .vercel/project.json après "vercel link"
#   VERCEL_PROJECT_ID: trouvable dans .vercel/project.json après "vercel link"

name: Deploy Web (Vercel)

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  deploy:
    name: Deploy to Vercel
    runs-on: ubuntu-latest
    timeout-minutes: 15

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Type check (web)
        run: npx turbo run type-check --filter=@immo-ci/web
        continue-on-error: false

      - name: Lint (web)
        run: npx turbo run lint --filter=@immo-ci/web
        continue-on-error: true    # lint ne bloque pas le déploiement

      - name: Build (web)
        run: npx turbo run build --filter=@immo-ci/web
        env:
          NEXT_PUBLIC_SUPABASE_URL: ${{ secrets.NEXT_PUBLIC_SUPABASE_URL }}
          NEXT_PUBLIC_SUPABASE_ANON_KEY: ${{ secrets.NEXT_PUBLIC_SUPABASE_ANON_KEY }}
          NEXT_PUBLIC_MAPBOX_TOKEN: ${{ secrets.NEXT_PUBLIC_MAPBOX_TOKEN }}
          NEXT_PUBLIC_CLOUDINARY_CLOUD_NAME: ${{ secrets.NEXT_PUBLIC_CLOUDINARY_CLOUD_NAME }}

      - name: Deploy to Vercel (Preview — PR)
        if: github.event_name == 'pull_request'
        run: npx vercel --token=${{ secrets.VERCEL_TOKEN }} --org-id=${{ secrets.VERCEL_ORG_ID }} --project-id=${{ secrets.VERCEL_PROJECT_ID }}
        env:
          VERCEL_TOKEN: ${{ secrets.VERCEL_TOKEN }}

      - name: Deploy to Vercel (Production — main)
        if: github.event_name == 'push' && github.ref == 'refs/heads/main'
        run: npx vercel --prod --token=${{ secrets.VERCEL_TOKEN }} --org-id=${{ secrets.VERCEL_ORG_ID }} --project-id=${{ secrets.VERCEL_PROJECT_ID }}
        env:
          VERCEL_TOKEN: ${{ secrets.VERCEL_TOKEN }}
```

**Fichier 2 — CD Mobile (`eas-build.yml`):**
Se déclenche sur push d'un tag `v*` (ex: `v1.0.0`). Build EAS production pour iOS + Android.

```yaml
# .github/workflows/eas-build.yml
# Build EAS production (iOS + Android) déclenché sur push de tag semver
# Usage: git tag v1.0.0 && git push --tags
#
# Secrets requis dans GitHub → Settings → Secrets and variables → Actions:
#   EXPO_TOKEN: depuis expo.dev/settings/access-tokens
#
# Prérequis:
#   - apps/mobile/app.json: extra.eas.projectId renseigné
#   - apps/mobile/eas.json: profil production configuré
#   - Certificats iOS gérés par EAS (auto-provisioned)
#   - Keystore Android géré par EAS (auto-generated premier build)

name: EAS Build (Mobile Production)

on:
  push:
    tags:
      - 'v[0-9]+.[0-9]+.[0-9]+'   # semver: v1.0.0, v1.2.3, etc.

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: false    # Ne pas annuler les builds en cours

jobs:
  build:
    name: EAS Build Production
    runs-on: ubuntu-latest
    timeout-minutes: 60         # Les builds EAS peuvent prendre 30-45 min

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'

      - name: Install dependencies (root)
        run: npm ci

      - name: Install EAS CLI
        run: npm install --global eas-cli

      - name: EAS Build — Android (App Bundle)
        working-directory: apps/mobile
        run: eas build --platform android --profile production --non-interactive --no-wait
        env:
          EXPO_TOKEN: ${{ secrets.EXPO_TOKEN }}

      - name: EAS Build — iOS
        working-directory: apps/mobile
        run: eas build --platform ios --profile production --non-interactive --no-wait
        env:
          EXPO_TOKEN: ${{ secrets.EXPO_TOKEN }}

      - name: Notify build submitted
        run: |
          echo "Builds EAS soumis pour le tag ${{ github.ref_name }}"
          echo "Suivre la progression sur: https://expo.dev/accounts/immo-ci/projects/immo-ci/builds"
```
