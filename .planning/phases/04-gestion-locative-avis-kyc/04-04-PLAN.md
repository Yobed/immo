**2. Créer `apps/web/components/kyc/KYCUploader.tsx`**

Upload CNI + selfie. Client Component. Utilise react-dropzone (déjà installé — utilisé par MediaUploader). Upload vers bucket 'kyc' Supabase Storage, puis appelle `/api/kyc` pour mettre à jour les URLs et le statut.

```typescript
// apps/web/components/kyc/KYCUploader.tsx
'use client'
import { useState, useCallback } from 'react'
import { useDropzone }           from 'react-dropzone'
import { createClient }          from '@/lib/supabase/client'
import { cn }                    from '@/lib/utils'

type KycDocType = 'cni' | 'selfie'

interface KYCUploaderProps {
  userId:         string
  currentCniUrl?: string    // URL existante si déjà uploadé
  currentSelfieUrl?: string
  onComplete?:    () => void
}

interface UploadState {
  url:       string | null
  uploading: boolean
  error:     string | null
}

function DocSlot({
  type, label, description, state,
  onDrop, currentUrl,
}: {
  type: KycDocType
  label: string
  description: string
  state: UploadState
  onDrop: (files: File[], type: KycDocType) => void
  currentUrl?: string
}) {
  const { getRootProps, getInputProps, isDragActive } = useDropzone({
    accept: { 'image/*': ['.jpg', '.jpeg', '.png'] },
    maxSize: 5 * 1024 * 1024,  // 5MB
    multiple: false,
    onDrop: files => onDrop(files, type),
    disabled: state.uploading,
  })

  const uploaded = state.url ?? currentUrl

  return (
    <div className="space-y-2">
      <label className="block text-sm font-semibold text-gray-700">{label}</label>
      <p className="text-xs text-gray-500">{description}</p>

      {uploaded ? (
        <div className="flex items-center gap-3 p-3 bg-green-50 border border-green-200 rounded-lg">
          <span className="text-green-600 text-lg">✓</span>
          <div>
            <p className="text-sm font-medium text-green-700">Document uploadé</p>
            <a href={uploaded} target="_blank" rel="noopener noreferrer"
               className="text-xs text-green-600 hover:underline">
              Voir le document
            </a>
          </div>
        </div>
      ) : (
        <div
          {...getRootProps()}
          className={cn(
            'border-2 border-dashed rounded-lg p-6 text-center cursor-pointer transition-colors',
            isDragActive ? 'border-primary bg-blue-50' : 'border-gray-200 hover:border-primary/50',
            state.uploading && 'opacity-50 cursor-not-allowed'
          )}
        >
          <input {...getInputProps()} />
          {state.uploading ? (
            <p className="text-sm text-gray-500">Upload en cours...</p>
          ) : (
            <>
              <p className="text-sm font-medium text-gray-700">Glisser-déposer ou cliquer</p>
              <p className="text-xs text-gray-400 mt-1">JPG, PNG · Max 5 MB</p>
            </>
          )}
          {state.error && (
            <p className="text-xs text-red-600 mt-2">{state.error}</p>
          )}
        </div>
      )}
    </div>
  )
}

export function KYCUploader({ userId, currentCniUrl, currentSelfieUrl, onComplete }: KYCUploaderProps) {
  const [cniState,    setCniState]    = useState<UploadState>({ url: null, uploading: false, error: null })
  const [selfieState, setSelfieState] = useState<UploadState>({ url: null, uploading: false, error: null })
  const [submitted,   setSubmitted]   = useState(false)
  const [submitError, setSubmitError] = useState<string | null>(null)

  const uploadDoc = useCallback(async (files: File[], type: KycDocType) => {
    if (!files[0]) return
    const setState = type === 'cni' ? setCniState : setSelfieState
    setState(s => ({ ...s, uploading: true, error: null }))

    const supabase = createClient()
    const file = files[0]
    const ext  = file.name.split('.').pop() ?? 'jpg'
    const path = `${userId}/${type}-${Date.now()}.${ext}`

    const { data, error: uploadError } = await supabase.storage
      .from('kyc')
      .upload(path, file, { cacheControl: '3600', upsert: true })

    if (uploadError || !data) {
      setState(s => ({ ...s, uploading: false, error: uploadError?.message ?? 'Erreur upload' }))
      return
    }

    // URL signée 1 an (bucket kyc est PRIVATE)
    const { data: signedData } = await supabase.storage
      .from('kyc')
      .createSignedUrl(data.path, 3600 * 24 * 365)

    const signedUrl = signedData?.signedUrl ?? ''
    setState({ url: signedUrl, uploading: false, error: null })

    // Appel API pour persister l'URL en base
    await fetch('/api/kyc', {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify({ type, url: signedUrl }),
    })
  }, [userId])

  const cniDone    = cniState.url    ?? currentCniUrl
  const selfieDone = selfieState.url ?? currentSelfieUrl
  const bothDone   = !!(cniDone && selfieDone)

  async function handleSubmitKyc() {
    if (!bothDone) return
    setSubmitError(null)

    const res = await fetch('/api/kyc', {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify({ submitKyc: true }),
    })

    if (!res.ok) {
      const data = await res.json()
      setSubmitError(data.error ?? 'Erreur')
      return
    }

    setSubmitted(true)
    onComplete?.()
  }

  if (submitted) {
    return (
      <div className="p-4 bg-yellow-50 border border-yellow-200 rounded-lg text-yellow-700 text-sm">
        <p className="font-semibold">Documents soumis pour vérification</p>
        <p className="mt-1">Notre équipe vérifiera vos documents sous 24-48h ouvrées.</p>
      </div>
    )
  }

  return (
    <div className="space-y-6">
      <DocSlot
        type="cni"
        label="Pièce d'identité (CNI)"
        description="Carte Nationale d'Identité recto-verso ou passeport"
        state={cniState}
        onDrop={uploadDoc}
        currentUrl={currentCniUrl}
      />
      <DocSlot
        type="selfie"
        label="Selfie avec votre CNI"
        description="Photo de vous tenant votre pièce d'identité"
        state={selfieState}
        onDrop={uploadDoc}
        currentUrl={currentSelfieUrl}
      />

      {bothDone && (
        <div className="pt-2">
          {submitError && <p className="text-sm text-red-600 mb-2">{submitError}</p>}
          <button
            onClick={handleSubmitKyc}
            className="w-full bg-primary text-white py-2.5 rounded-lg font-medium hover:bg-primary/90 transition-colors"
          >
            Soumettre pour vérification
          </button>
        </div>
      )}
    </div>
  )
}
```

**3. Créer `apps/web/app/api/kyc/route.ts`**

Gère deux actions: (a) POST individuel avec `{type, url}` pour persister une URL de document, (b) POST avec `{submitKyc:true}` pour passer kyc_statut à 'en_cours', (c) PATCH avec `{statut}` pour validation admin.

```typescript
// apps/web/app/api/kyc/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { createClient }              from '@/lib/supabase/server'

export async function POST(req: NextRequest) {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Non authentifié' }, { status: 401 })

  const body = await req.json() as {
    type?:      'cni' | 'selfie'
    url?:       string
    submitKyc?: boolean
  }

  // Action 1: soumettre pour vérification (passer en_cours)
  if (body.submitKyc) {
    // Vérifier que les deux documents sont uploadés
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const { data: profile } = await (supabase.from('profiles') as any)
      .select('kyc_cni_url, kyc_selfie_url, kyc_statut')
      .eq('id', user.id)
      .single()

    if (!profile?.kyc_cni_url || !profile?.kyc_selfie_url) {
      return NextResponse.json({ error: 'Les deux documents (CNI et selfie) doivent être uploadés' }, { status: 400 })
    }
    if (profile.kyc_statut === 'verifie') {
      return NextResponse.json({ error: 'Votre identité est déjà vérifiée' }, { status: 409 })
    }

    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    await (supabase.from('profiles') as any)
      .update({ kyc_statut: 'en_cours' })
      .eq('id', user.id)

    return NextResponse.json({ success: true, statut: 'en_cours' })
  }

  // Action 2: persister URL d'un document
  if (!body.type || !body.url) {
    return NextResponse.json({ error: 'type et url requis' }, { status: 400 })
  }
  if (!['cni', 'selfie'].includes(body.type)) {
    return NextResponse.json({ error: 'type doit être cni ou selfie' }, { status: 400 })
  }

  const field = body.type === 'cni' ? 'kyc_cni_url' : 'kyc_selfie_url'

  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  const { error: updateError } = await (supabase.from('profiles') as any)
    .update({ [field]: body.url })
    .eq('id', user.id)

  if (updateError) {
    return NextResponse.json({ error: updateError.message }, { status: 500 })
  }

  return NextResponse.json({ success: true, field, url: body.url })
}

// PATCH: validation admin — change kyc_statut vers 'verifie' ou 'non_verifie'
export async function PATCH(req: NextRequest) {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Non authentifié' }, { status: 401 })

  const body = await req.json() as { userId: string; statut: 'verifie' | 'non_verifie' | 'en_cours' }

  // TODO Phase 5: vérifier que l'utilisateur est admin (role='admin')
  // Pour Phase 4, seul un utilisateur peut modifier son propre profil via RLS
  const targetId = body.userId ?? user.id

  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  await (supabase.from('profiles') as any)
    .update({ kyc_statut: body.statut })
    .eq('id', targetId)

  // Si kyc_valide: insérer notification
  if (body.statut === 'verifie') {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    await (supabase.from('notifications') as any).insert({
      user_id: targetId,
      type:    'kyc_valide',
      titre:   'Identité vérifiée',
      contenu: 'Votre identité a été vérifiée avec succès. Votre profil est maintenant certifié.',
    })
  }

  return NextResponse.json({ success: true })
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { createClient }              from '@/lib/supabase/server'

export async function GET(_req: NextRequest) {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Non authentifié' }, { status: 401 })

  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  const { data: notifications, error } = await (supabase.from('notifications') as any)
    .select('id, type, titre, contenu, lu, lien_type, lien_id, created_at')
    .eq('user_id', user.id)
    .order('created_at', { ascending: false })
    .limit(50)

  if (error) return NextResponse.json({ error: error.message }, { status: 500 })

  const unreadCount = (notifications ?? []).filter((n: any) => !n.lu).length

  return NextResponse.json({ notifications: notifications ?? [], unreadCount })
}
```

```typescript
// apps/web/app/api/notifications/[id]/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { createClient }              from '@/lib/supabase/server'

export async function PATCH(
  _req: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Non authentifié' }, { status: 401 })

  const { id } = await params

  // RLS "Utilisateur marque ses notifications lues" garantit user_id = auth.uid()
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  const { error } = await (supabase.from('notifications') as any)
    .update({ lu: true })
    .eq('id', id)
    .eq('user_id', user.id)  // double sécurité

  if (error) return NextResponse.json({ error: error.message }, { status: 500 })

  return NextResponse.json({ success: true })
}
```

```typescript
import { useState, useCallback }    from 'react'
import { NotificationItem }         from './NotificationItem'
import type { NotificationData }    from './NotificationItem'

interface NotificationCenterProps {
  initialNotifications: NotificationData[]
}

export function NotificationCenter({ initialNotifications }: NotificationCenterProps) {
  const [notifications, setNotifications] = useState<NotificationData[]>(initialNotifications)

  const markRead = useCallback(async (id: string) => {
    // Optimistic update
    setNotifications(prev =>
      prev.map(n => n.id === id ? { ...n, lu: true } : n)
    )
    await fetch(`/api/notifications/${id}`, { method: 'PATCH' })
  }, [])

  const markAllRead = useCallback(async () => {
    const unread = notifications.filter(n => !n.lu)
    // Optimistic update
    setNotifications(prev => prev.map(n => ({ ...n, lu: true })))
    await Promise.all(unread.map(n =>
      fetch(`/api/notifications/${n.id}`, { method: 'PATCH' })
    ))
  }, [notifications])

  const unreadCount = notifications.filter(n => !n.lu).length

            </span>
          )}
        </h1>
        {unreadCount > 0 && (
          <button
            onClick={markAllRead}
            className="text-sm text-primary hover:underline"
          >
```

```typescript
// apps/web/components/notifications/NotificationBell.tsx
'use client'
import { useState, useEffect, useCallback, useRef } from 'react'
import { createClient }                               from '@/lib/supabase/client'
import { NotificationItem }                           from './NotificationItem'
import type { NotificationData }                      from './NotificationItem'
import { cn }                                         from '@/lib/utils'

interface NotificationBellProps {
  userId:              string
  initialUnreadCount?: number
}

export function NotificationBell({ userId, initialUnreadCount = 0 }: NotificationBellProps) {
  const [notifications, setNotifications] = useState<NotificationData[]>([])
  const [unreadCount,   setUnreadCount]   = useState(initialUnreadCount)
  const [open,          setOpen]          = useState(false)
  const [loaded,        setLoaded]        = useState(false)
  const dropdownRef = useRef<HTMLDivElement>(null)

  // Charger les notifications au clic (lazy load)
  const loadNotifications = useCallback(async () => {
    if (loaded) return
    const res  = await fetch('/api/notifications')
    const data = await res.json()
    setNotifications(data.notifications ?? [])
    setUnreadCount(data.unreadCount ?? 0)
    setLoaded(true)
  }, [loaded])

  // Supabase Realtime — écouter les nouvelles notifications
  useEffect(() => {
    const supabase = createClient()
    const channel  = supabase
      .channel(`notifications-${userId}`)
      .on(
        'postgres_changes',
        {
          event:  'INSERT',
          schema: 'public',
          table:  'notifications',
          filter: `user_id=eq.${userId}`,
        },
        (payload) => {
          const newNotif = payload.new as NotificationData
          setNotifications(prev => [newNotif, ...prev])
          setUnreadCount(c => c + 1)
        }
      )
      .subscribe()

    // CRITIQUE: cleanup obligatoire pour éviter les fuites mémoire
    return () => { supabase.removeChannel(channel) }
  }, [userId])

  // Fermer le dropdown au clic extérieur
  useEffect(() => {
    function handleClickOutside(e: MouseEvent) {
      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
        setOpen(false)
      }
    }
    if (open) document.addEventListener('mousedown', handleClickOutside)
    return () => document.removeEventListener('mousedown', handleClickOutside)
  }, [open])

  const markRead = useCallback(async (id: string) => {
    setNotifications(prev => prev.map(n => n.id === id ? { ...n, lu: true } : n))
    setUnreadCount(c => Math.max(0, c - 1))
    await fetch(`/api/notifications/${id}`, { method: 'PATCH' })
  }, [])

  const handleBellClick = () => {
    setOpen(prev => !prev)
    if (!open) loadNotifications()
  }

  return (
    <div ref={dropdownRef} className="relative">
      {/* Bouton cloche */}
      <button
        onClick={handleBellClick}
        className="relative p-2 text-gray-600 hover:text-gray-900 transition-colors"
        aria-label={`Notifications${unreadCount > 0 ? ` — ${unreadCount} non lues` : ''}`}
      >
        <svg width="22" height="22" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
          <path d="M18 8A6 6 0 0 0 6 8c0 7-3 9-3 9h18s-3-2-3-9" />
          <path d="M13.73 21a2 2 0 0 1-3.46 0" />
        </svg>
        {unreadCount > 0 && (
          <span className="absolute -top-0.5 -right-0.5 min-w-[18px] h-[18px] bg-red-500 text-white text-[10px] font-bold rounded-full flex items-center justify-center px-1">
            {unreadCount > 99 ? '99+' : unreadCount}
          </span>
        )}
      </button>

      {/* Dropdown */}
      {open && (
        <div className={cn(
          'absolute right-0 top-full mt-2 w-80 bg-white border border-gray-200 rounded-xl shadow-xl z-50',
          'max-h-[420px] overflow-y-auto'
        )}>
          <div className="p-3 border-b border-gray-100 flex items-center justify-between">
            <span className="text-sm font-semibold text-gray-800">Notifications</span>
            {unreadCount > 0 && (
              <span className="text-xs text-primary">
                {unreadCount} non lu{unreadCount > 1 ? 'es' : 'e'}
              </span>
            )}
          </div>

          {!loaded ? (
            <div className="p-4 text-center text-sm text-gray-500">Chargement...</div>
          ) : notifications.length === 0 ? (
            <div className="p-6 text-center text-sm text-gray-500">Aucune notification</div>
          ) : (
            notifications.slice(0, 10).map(n => (
              <NotificationItem key={n.id} notification={n} onMarkRead={markRead} />
            ))
          )}

          <div className="p-2 border-t border-gray-100 text-center">
            <a href="/notifications" className="text-xs text-primary hover:underline">
              Voir toutes les notifications
            </a>
          </div>
        </div>
      )}
    </div>
  )
}
```
